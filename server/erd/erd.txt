# таблица пользователей, пароль хранится как bcrypt-хэш
users
---
user_id serial PK
username text unique
password text # bcrypt-хэш

# таблица контактов, то как пользователи называют
# друг друга, кто с кем знаком и т.д.
contacts
---
user_1 serial PK FK >- users.user_id
user_2 serial PK FK >- users.user_id
nickname varchar(255)

# таблица чатов и их типы
chats
---
chat_id serial PK
title varchar(255)
chat_type chat_type_enum # Private, Group, Channel

participants
---
chat_id serial PK FK >- chats.chat_id
user_id serial PK FK >- users.user_id

roles
---
chat_id serial PK FK >- chats.chat_id
user_id serial PK FK >- users.user_id
role_type role_type_enum # Owner, Administrator, Participant

messages
---
message_id serial PK
chat_id serial FK >- chats.chat_id
created_at timestamp
modified_at timestamp # выставляется триггером
message_text text

voices
---
voice_id serial PK
message_id serial FK >- messages.message_id
media_url text
waveform varchar(255) # csv значения, формируется триггером

images
---
image_id serial PK
message_id serial FK >- messages.message_id
media_url text
thumbnail text # base64 картинка для предпросмотра

files
---
file_id serial PK
message_id serial FK >- messages.message_id
media_url text
